CREATE TEMP TABLE telemetria_embotellado AS
SELECT
    lata_id,
    CASE
        WHEN random() < 0.10 THEN NULL
        WHEN random() BETWEEN 0.10 AND 0.15 THEN ROUND(420 + (random() * 60), 2)
        WHEN random() > 0.95 THEN ROUND(200 + (random() * 50), 2)
        ELSE ROUND(350 + (random() * 10), 2)
    END AS volumen_ml
FROM generate_series(1, 100) AS t(lata_id);

WITH mediana_calculada AS (
    SELECT MEDIAN(volumen_ml) AS val_mediana
    FROM telemetria_embotellado
)
SELECT
    t.lata_id,
    t.volumen_ml AS valor_original,
    COALESCE(t.volumen_ml, m.val_mediana) AS volumen_imputado
FROM telemetria_embotellado t, mediana_calculada m
ORDER BY t.lata_id;

WITH paso1_imputacion AS (
    SELECT
        lata_id,
        COALESCE(volumen_ml, (SELECT MEDIAN(volumen_ml) FROM telemetria_embotellado)) AS volumen_imputado
    FROM telemetria_embotellado
),
paso2_metricas AS (
    SELECT
        quantile_cont(volumen_imputado, 0.25) AS Q1,
        quantile_cont(volumen_imputado, 0.75) AS Q3,
        (quantile_cont(volumen_imputado, 0.75) - quantile_cont(volumen_imputado, 0.25)) AS IQR
    FROM paso1_imputacion
),
paso3_limites AS (
    SELECT
        (Q1 - 1.5 * IQR) AS limite_inf,
        (Q3 + 1.5 * IQR) AS limite_sup
    FROM paso2_metricas
)
SELECT
    ROUND((SELECT MEDIAN(volumen_ml) FROM telemetria_embotellado), 2) AS mediana_imputacion,
    ROUND(m.Q1, 2) AS Q1,
    ROUND(m.Q3, 2) AS Q3,
    ROUND(m.IQR, 2) AS IQR,
    ROUND(l.limite_inf, 2) AS limite_inferior,
    ROUND(l.limite_sup, 2) AS limite_superior
FROM paso2_metricas m, paso3_limites l;
